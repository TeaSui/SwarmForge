# Plan 02-01: SQS Consumer

**Phase:** 2 — Orchestration & LLM
**Wave:** 1 (no dependencies — foundational piece for Phase 2)
**Requirements:** ORC-03

## Goal

Implement a long-polling SQS consumer that runs as a background thread alongside the FastAPI server. It picks up task messages from the queue, dispatches them to the orchestrator, and deletes messages only on success (letting visibility timeout handle retries).

## must_haves

- `consumer/sqs_consumer.py` long-polls SQS with `WaitTimeSeconds=20`
- `VisibilityTimeout=1800` (30 min) — long enough for a full agent run
- Messages dispatched to `orchestrator.dispatch(body)` (stub in this plan — wired in 02-02)
- `delete_message` called ONLY after successful dispatch
- Consumer started as daemon thread when FastAPI app starts (via `lifespan`)
- Tests use `moto` for SQS mocking

<task id="1">
Create `consumer/sqs_consumer.py`:

```python
import boto3, json, threading, logging
from botocore.exceptions import ClientError

logger = logging.getLogger(__name__)

class SQSConsumer:
    def __init__(self, queue_url: str, region: str, orchestrator):
        self.queue_url = queue_url
        self.region = region
        self.orchestrator = orchestrator
        self.sqs = boto3.client("sqs", region_name=region)
        self.running = False
        self._thread: threading.Thread | None = None

    def start(self):
        self.running = True
        self._thread = threading.Thread(target=self._poll_loop, daemon=True, name="sqs-consumer")
        self._thread.start()
        logger.info("SQS consumer started")

    def stop(self):
        self.running = False
        if self._thread:
            self._thread.join(timeout=5)
        logger.info("SQS consumer stopped")

    def _poll_loop(self):
        while self.running:
            try:
                self._poll_once()
            except Exception as e:
                logger.error(f"SQS poll error: {e}", exc_info=True)

    def _poll_once(self):
        response = self.sqs.receive_message(
            QueueUrl=self.queue_url,
            MaxNumberOfMessages=1,
            WaitTimeSeconds=20,
            VisibilityTimeout=1800,
            MessageAttributeNames=["All"],
        )
        for msg in response.get("Messages", []):
            self._handle_message(msg)

    def _handle_message(self, msg: dict):
        msg_id = msg["MessageId"]
        receipt = msg["ReceiptHandle"]
        try:
            body = json.loads(msg["Body"])
            logger.info(f"Processing task {body.get('task_id', '?')} from {body.get('source', '?')}")
            self.orchestrator.dispatch(body)
            self._delete_message(receipt)
        except Exception as e:
            logger.error(f"Failed to process message {msg_id}: {e}", exc_info=True)
            # Do NOT delete — let visibility timeout expire → DLQ after 3 attempts

    def _delete_message(self, receipt_handle: str):
        self.sqs.delete_message(QueueUrl=self.queue_url, ReceiptHandle=receipt_handle)
```

<task id="2">
Wire consumer into FastAPI app lifecycle (`main.py`):

```python
from contextlib import asynccontextmanager
from consumer.sqs_consumer import SQSConsumer
from config import settings

# Stub orchestrator for Phase 2 plan 02-01 (real one wired in 02-02)
class _StubOrchestrator:
    def dispatch(self, body: dict):
        import logging
        logging.getLogger(__name__).info(f"[STUB] Would dispatch task: {body.get('task_id')}")

_consumer = SQSConsumer(
    queue_url=settings.SQS_QUEUE_URL,
    region=settings.AWS_REGION,
    orchestrator=_StubOrchestrator(),
)

@asynccontextmanager
async def lifespan(app: FastAPI):
    _consumer.start()
    yield
    _consumer.stop()

app = FastAPI(title="SwarmForge Ingress", version="0.1.0", lifespan=lifespan)
```

```python
import pytest, json, boto3
from moto import mock_aws
from unittest.mock import MagicMock
from consumer.sqs_consumer import SQSConsumer

@pytest.fixture
def aws_credentials(monkeypatch):
    monkeypatch.setenv("AWS_ACCESS_KEY_ID", "testing")
    monkeypatch.setenv("AWS_SECRET_ACCESS_KEY", "testing")
    monkeypatch.setenv("AWS_DEFAULT_REGION", "ap-southeast-1")

@pytest.fixture
def sqs_setup(aws_credentials):
    with mock_aws():
        client = boto3.client("sqs", region_name="ap-southeast-1")
        q = client.create_queue(QueueName="test-tasks")
        yield client, q["QueueUrl"]

def send_test_message(client, url, task_id="task-1"):
    client.send_message(
        QueueUrl=url,
        MessageBody=json.dumps({"task_id": task_id, "source": "jira", "event_type": "test"}),
    )

def test_message_dispatched_and_deleted(sqs_setup):
    client, url = sqs_setup
    mock_orchestrator = MagicMock()
    consumer = SQSConsumer(queue_url=url, region="ap-southeast-1", orchestrator=mock_orchestrator)

    send_test_message(client, url)
    consumer._poll_once()

    mock_orchestrator.dispatch.assert_called_once()
    # Message should be deleted
    remaining = client.receive_message(QueueUrl=url).get("Messages", [])
    assert len(remaining) == 0

def test_failed_dispatch_does_not_delete(sqs_setup):
    client, url = sqs_setup
    mock_orchestrator = MagicMock()
    mock_orchestrator.dispatch.side_effect = RuntimeError("Simulated crash")
    consumer = SQSConsumer(queue_url=url, region="ap-southeast-1", orchestrator=mock_orchestrator)

    # Temporarily lower visibility for test
    client.send_message(QueueUrl=url, MessageBody=json.dumps({"task_id": "f", "source": "jira"}))
    consumer._poll_once()

    # Message should NOT be deleted (will reappear after visibility timeout)
    mock_orchestrator.dispatch.assert_called_once()
```
