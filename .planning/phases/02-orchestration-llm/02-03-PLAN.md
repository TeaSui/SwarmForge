# Plan 02-03: LLM Layer — Claude + Bedrock Failover

**Phase:** 2 — Orchestration & LLM
**Wave:** 2 (parallel with 02-02 and 02-04)
**Requirements:** LLM-01, LLM-02, LLM-03, LLM-04

## Goal

Implement a unified LLM client that calls Anthropic Claude API as primary, automatically fails over to AWS Bedrock on rate limit or 5xx errors, and enables prompt caching for repeated large contexts. This client is used by all agents in Phase 3.

## must_haves

- `llm/client.py` — `LLMClient.chat(messages, system)` works transparently with both providers
- Auto-failover on `RateLimitError` and `APIStatusError` (5xx) — not on 4xx client errors
- Prompt caching via `cache_control: {type: ephemeral}` on stable system context blocks
- LangSmith tracing wrapper logs every LLM call (model, tokens, latency)
- Client initialized as singleton using settings from env/Secrets Manager

```python
# llm/client.py
import json, logging, time
from anthropic import Anthropic, RateLimitError, APIStatusError
import boto3
from config import settings

logger = logging.getLogger(__name__)

class LLMClient:
    PRIMARY_MODEL = "claude-3-5-sonnet-20241022"
    FALLBACK_MODEL = "global.anthropic.claude-sonnet-4-6"

    def __init__(self):
        self.anthropic = Anthropic(api_key=settings.ANTHROPIC_API_KEY)
        self.bedrock = boto3.client("bedrock-runtime", region_name=settings.AWS_REGION)
        self._failover_count = 0

    def chat(
        self,
        messages: list[dict],
        system: str = "",
        max_tokens: int = 4096,
        cache_system: bool = False,
    ) -> tuple[str, dict]:
        """Returns (text, usage_dict)."""
        try:
            return self._call_anthropic(messages, system, max_tokens, cache_system)
        except (RateLimitError, APIStatusError) as e:
            if isinstance(e, APIStatusError) and e.status_code < 500:
                raise  # 4xx = client error, don't failover
            self._failover_count += 1
            logger.warning(f"Claude API error ({type(e).__name__}), failing over to Bedrock (total failovers: {self._failover_count})")
            return self._call_bedrock(messages, system, max_tokens)

    def _call_anthropic(self, messages, system, max_tokens, cache_system) -> tuple[str, dict]:
        start = time.monotonic()
        sys_content = (
            [{"type": "text", "text": system, "cache_control": {"type": "ephemeral"}}]
            if cache_system and system
            else system
        )
        resp = self.anthropic.messages.create(
            model=self.PRIMARY_MODEL,
            messages=messages,
            system=sys_content,
            max_tokens=max_tokens,
        )
        latency_ms = int((time.monotonic() - start) * 1000)
        usage = {
            "input_tokens": resp.usage.input_tokens,
            "output_tokens": resp.usage.output_tokens,
            "total_tokens": resp.usage.input_tokens + resp.usage.output_tokens,
            "model": self.PRIMARY_MODEL,
            "latency_ms": latency_ms,
        }
        logger.info(f"Claude: {usage['total_tokens']} tokens, {latency_ms}ms")
        return resp.content[0].text, usage

    def _call_bedrock(self, messages, system, max_tokens) -> tuple[str, dict]:
        start = time.monotonic()
        body = json.dumps({
            "anthropic_version": "bedrock-2023-05-31",
            "messages": messages,
            "system": system,
            "max_tokens": max_tokens,
        })
        response = self.bedrock.invoke_model(
            modelId=self.FALLBACK_MODEL,
            body=body,
            contentType="application/json",
            accept="application/json",
        )
        result = json.loads(response["body"].read())
        latency_ms = int((time.monotonic() - start) * 1000)
        usage = {
            "input_tokens": result["usage"]["input_tokens"],
            "output_tokens": result["usage"]["output_tokens"],
            "total_tokens": result["usage"]["input_tokens"] + result["usage"]["output_tokens"],
            "model": self.FALLBACK_MODEL,
            "latency_ms": latency_ms,
        }
        logger.info(f"Bedrock: {usage['total_tokens']} tokens, {latency_ms}ms")
        return result["content"][0]["text"], usage

# Singleton
llm_client = LLMClient()
```

```python
import pytest
from unittest.mock import MagicMock, patch
from anthropic import RateLimitError, APIStatusError

def test_primary_call_returns_text(monkeypatch):
    with patch("llm.client.Anthropic") as mock_anthropic:
        mock_resp = MagicMock()
        mock_resp.content = [MagicMock(text="Hello world")]
        mock_resp.usage = MagicMock(input_tokens=10, output_tokens=5)
        mock_anthropic.return_value.messages.create.return_value = mock_resp
        from llm.client import LLMClient
        client = LLMClient()
        text, usage = client.chat([{"role": "user", "content": "hi"}])
        assert text == "Hello world"
        assert usage["total_tokens"] == 15

def test_failover_on_rate_limit(monkeypatch):
    with patch("llm.client.Anthropic") as mock_anthropic, \
         patch("llm.client.boto3") as mock_boto3:
        mock_anthropic.return_value.messages.create.side_effect = RateLimitError(
            message="rate limit", response=MagicMock(status_code=429), body={}
        )
        mock_bedrock = MagicMock()
        mock_boto3.client.return_value = mock_bedrock
        import json
        mock_bedrock.invoke_model.return_value = {
            "body": MagicMock(read=lambda: json.dumps({
                "content": [{"text": "bedrock response"}],
                "usage": {"input_tokens": 20, "output_tokens": 10}
            }).encode())
        }
        from llm.client import LLMClient
        client = LLMClient()
        text, usage = client.chat([{"role": "user", "content": "hi"}])
        assert text == "bedrock response"
        assert usage["model"] == LLMClient.FALLBACK_MODEL

def test_no_failover_on_4xx():
    """4xx errors should NOT trigger failover."""
    with patch("llm.client.Anthropic") as mock_anthropic:
        mock_anthropic.return_value.messages.create.side_effect = APIStatusError(
            message="bad request", response=MagicMock(status_code=400), body={}
        )
        from llm.client import LLMClient
        client = LLMClient()
        with pytest.raises(APIStatusError):
            client.chat([{"role": "user", "content": "hi"}])
```
