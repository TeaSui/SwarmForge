```python
import boto3, time
from datetime import datetime, UTC
from config import settings

class StateManager:
    def __init__(self):
        self.client = boto3.client("dynamodb", region_name=settings.AWS_REGION)
        self.table = settings.DYNAMODB_TASKS_TABLE  # "swarmforge-tasks"

    def create_task(self, task_id: str, body: dict) -> None:
        self.client.put_item(
            TableName=self.table,
            Item={
                "task_id": {"S": task_id},
                "source": {"S": body.get("source", "unknown")},
                "event_type": {"S": body.get("event_type", "unknown")},
                "jira_issue_id": {"S": body.get("payload", {}).get("issue", {}).get("id", "")},
                "status": {"S": "PENDING"},
                "created_at": {"S": datetime.now(UTC).isoformat()},
                "updated_at": {"S": datetime.now(UTC).isoformat()},
                "token_usage": {"M": {"input": {"N": "0"}, "output": {"N": "0"}, "total": {"N": "0"}}},
                "token_budget": {"N": str(settings.DEFAULT_TOKEN_BUDGET)},
                "ttl": {"N": str(int(time.time()) + 7 * 86400)},  # 7-day TTL
            },
        )

    def update_status(self, task_id: str, status: str, **extra):
        update_expr = "SET #s = :s, updated_at = :u"
        expr_names = {"#s": "status"}
        expr_values = {":s": {"S": status}, ":u": {"S": datetime.now(UTC).isoformat()}}
        for key, val in extra.items():
            update_expr += f", {key} = :{key}"
            expr_values[f":{key}"] = {"S": str(val)}
        self.client.update_item(
            TableName=self.table,
            Key={"task_id": {"S": task_id}},
            UpdateExpression=update_expr,
            ExpressionAttributeNames=expr_names,
            ExpressionAttributeValues=expr_values,
        )
```

```python
from config import settings

INTENT_KEYWORDS = {
    "test_generation": ["write test", "add test", "unit test", "integration test", "pytest"],
    "bug_fix": ["fix", "bug", "error", "broken", "failing", "crash", "exception"],
    "documentation": ["write doc", "update doc", "readme", "docstring", "document"],
    "code_review": ["review", "check code", "audit", "inspect"],
    "code_generation": ["implement", "create", "build", "add feature", "new endpoint"],
}

def parse_intent(payload: dict, llm_client=None) -> str:
    """Rule-based fast path, LLM fallback."""
    issue = payload.get("payload", {}).get("issue", {})
    summary = issue.get("fields", {}).get("summary", "").lower()
    description = str(issue.get("fields", {}).get("description", "")).lower()
    text = f"{summary} {description}"

    for intent, keywords in INTENT_KEYWORDS.items():
        if any(kw in text for kw in keywords):
            return intent

    # LLM fallback
    if llm_client:
        prompt = f"Classify this Jira ticket intent in one word from [code_generation, code_review, bug_fix, documentation, test_generation, unknown]:\n\nSummary: {summary}\nDescription: {description[:500]}\n\nIntent:"
        result = llm_client.chat([{"role": "user", "content": prompt}])
        intent = result.strip().lower().split()[0]
        return intent if intent in INTENT_KEYWORDS else "code_generation"

    return "code_generation"  # Safe default
```

import logging
from orchestrator.state_manager import StateManager
from orchestrator.intent_parser import parse_intent

logger = logging.getLogger(__name__)

class Orchestrator:
    def __init__(self, llm_client=None):
        self.state = StateManager()
        self.llm = llm_client

    def dispatch(self, body: dict):
        task_id = body["task_id"]
        logger.info(f"Dispatching task {task_id}")

        # 1. Persist PENDING state
        self.state.create_task(task_id, body)

        try:
            # 2. Parse intent
            intent = parse_intent(body, self.llm)
            self.state.update_status(task_id, "INTENT_PARSED", intent=intent)
            logger.info(f"Task {task_id} intent: {intent}")

            # 3. Mark IN_PROGRESS
            self.state.update_status(task_id, "IN_PROGRESS")

            # 4. Run crew (stubbed — real agents wired in Phase 3)
            result = self._run_crew(intent, body, task_id)

            # 5. Mark COMPLETE
            self.state.update_status(task_id, "COMPLETE", result=str(result)[:1000])
            logger.info(f"Task {task_id} completed")

        except Exception as e:
            logger.error(f"Task {task_id} failed: {e}", exc_info=True)
            self.state.update_status(task_id, "FAILED", error=str(e)[:500])
            raise

    def _run_crew(self, intent: str, body: dict, task_id: str) -> str:
        """Stub — Phase 3 wires real CrewAI crews here."""
        logger.info(f"[STUB] Running crew for intent={intent}, task_id={task_id}")
        return f"stub_result_for_{intent}"
