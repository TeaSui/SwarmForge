# Plan 01-03: Deduplication

**Phase:** 1 — Ingress & Queue
**Wave:** 2 (depends on Plan 01-01; runs in parallel with 01-02)
**Requirements:** TRG-06

## Goal

Implement idempotent event processing using DynamoDB conditional writes. Before publishing to SQS, each webhook route checks whether the event's idempotency key has already been processed within the current hour window. Duplicate events return 200 silently without re-publishing.

## must_haves

- `services/deduplication.py` checks DynamoDB before any SQS publish
- First occurrence of an idempotency key → writes to DynamoDB, returns `True` (proceed)
- Duplicate occurrence → DynamoDB conditional write fails → returns `False` (skip SQS)
- DynamoDB TTL set to 1 hour — duplicates are allowed after TTL expires
- All 3 webhook routes call dedup check before `background_tasks.add_task(...)`
- Tests use `moto` — no real AWS calls

<task id="1">
Implement `services/deduplication.py`:

```python
import boto3, time
from datetime import datetime, UTC
from botocore.exceptions import ClientError
import logging

logger = logging.getLogger(__name__)

class DynamoDBDeduplicator:
    def __init__(self, table_name: str, region: str):
        self.table_name = table_name
        self.client = boto3.client("dynamodb", region_name=region)
        self.ttl_seconds = 3600  # 1 hour

    def is_new_event(self, idempotency_key: str) -> bool:
        """
        Returns True if this is a NEW event (process it).
        Returns False if it's a duplicate (skip it).
        """
        try:
            self.client.put_item(
                TableName=self.table_name,
                Item={
                    "idempotency_key": {"S": idempotency_key},
                    "created_at": {"S": datetime.now(UTC).isoformat()},
                    "expires_at": {"N": str(int(time.time()) + self.ttl_seconds)},
                },
                ConditionExpression="attribute_not_exists(idempotency_key)",
            )
            logger.debug(f"New event: {idempotency_key[:16]}...")
            return True
        except ClientError as e:
            if e.response["Error"]["Code"] == "ConditionalCheckFailedException":
                logger.info(f"Duplicate event suppressed: {idempotency_key[:16]}...")
                return False
            # Unexpected DynamoDB error — log and allow through (fail open)
            logger.error(f"DynamoDB dedup check failed: {e}. Allowing event through.")
            return True

# Singleton
from config import settings
deduplicator = DynamoDBDeduplicator(
    table_name=settings.DYNAMODB_IDEMPOTENCY_TABLE,
    region=settings.AWS_REGION,
)
```

Update `api/webhooks/jira.py` to call dedup before SQS publish:

```python
from services.deduplication import deduplicator
from services.sqs_producer import sqs_producer

@router.post("/jira")
async def jira_webhook(request: Request, background_tasks: BackgroundTasks):
    body = await request.body()
    _validate_jira_signature(body, request.headers.get("X-Hub-Signature-256"))
    payload = await request.json()

    if not _is_ai_task(payload):
        return {"status": "accepted"}  # Not an AI task, skip silently

    idempotency_key = sqs_producer._build_idempotency_key(payload, "jira")
    if not deduplicator.is_new_event(idempotency_key):
        return {"status": "accepted"}  # Duplicate, skip silently

    background_tasks.add_task(sqs_producer.publish, payload, "jira")
    return {"status": "accepted"}
```

from moto import mock_aws
from services.deduplication import DynamoDBDeduplicator

@pytest.fixture

def test_first_event_is_new(dynamo_table):

def test_second_event_is_duplicate(dynamo_table):

    assert dedup.is_new_event("key-xyz-456") is True   # First time

    assert dedup.is_new_event("key-beta") is True  # Different key = new

def test_dedup_full_webhook_flow(dynamo_table, monkeypatch):
    """Integration: same Jira event sent twice → second is deduplicated."""
    # Import after fixture setup

    import hmac, hashlib, json, time
    from main import app
    from config import settings

                        DynamoDBDeduplicator(table_name=dynamo_table, region="ap-southeast-1"))

        "issue": {"id": "DUP-001", "fields": {"labels": [{"name": "ai-task"}]}},

    }

    r2 = client.post("/webhook/jira", content=body, headers=headers)

    assert r2.status_code == 200  # Both return 200 — dedup is silent
