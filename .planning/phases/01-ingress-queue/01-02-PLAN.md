**Phase:** 1 — Ingress & Queue

**Requirements:** TRG-04, TRG-05

## Goal

Implement the SQS producer service that serializes trigger payloads into a standardized task message envelope and publishes them to the SQS queue. Wire it into all three webhook routes (replacing the stubs from 01-01). Configure DLQ parameters in service code (CDK sets up actual infra in Phase 5).

## must_haves

- `services/sqs_producer.py` publishes a valid JSON message envelope to SQS for every AI-labelled event
- Message envelope includes: `task_id` (UUID), `source`, `event_type`, `idempotency_key`, `received_at`, `payload`
- All 3 webhook routes (`/webhook/jira`, `/webhook/slack`, `/webhook/generic`) call the real SQS producer via `BackgroundTasks`
- SQS client uses `boto3` with env-configured `SQS_QUEUE_URL`
- Tests use `moto` to mock SQS — no real AWS calls

Implement `services/sqs_producer.py`:

```python
import boto3, json, uuid, hashlib, time
from datetime import datetime, UTC
from botocore.exceptions import ClientError
import logging

logger = logging.getLogger(__name__)

class SQSProducer:
    def __init__(self, queue_url: str, region: str):
        self.queue_url = queue_url
        self.client = boto3.client("sqs", region_name=region)

    def _build_idempotency_key(self, payload: dict, source: str) -> str:
        if source == "jira":
            issue_id = payload.get("issue", {}).get("id", "")
            event_type = payload.get("webhookEvent", "")
            ts = payload.get("timestamp", 0)
            hour_bucket = str(int(ts) // 3_600_000)  # Jira ts in ms
            raw = f"jira:{issue_id}:{event_type}:{hour_bucket}"
        elif source == "slack":
            raw = f"slack:{payload.get('event_id', uuid.uuid4())}"
        else:
            raw = f"generic:{payload.get('id', uuid.uuid4())}"
        return hashlib.sha256(raw.encode()).hexdigest()

    def publish(self, payload: dict, source: str) -> str | None:
        idempotency_key = self._build_idempotency_key(payload, source)
        message = {
            "task_id": str(uuid.uuid4()),
            "source": source,
            "event_type": self._extract_event_type(payload, source),
            "idempotency_key": idempotency_key,
            "received_at": datetime.now(UTC).isoformat(),
            "payload": payload,
        }
        try:
            response = self.client.send_message(
                QueueUrl=self.queue_url,
                MessageBody=json.dumps(message),
                MessageAttributes={
                    "source": {"DataType": "String", "StringValue": source},
                    "task_id": {"DataType": "String", "StringValue": message["task_id"]},
                },
            )
            logger.info(f"Published task {message['task_id']} from {source} → SQS MessageId={response['MessageId']}")
            return response["MessageId"]
        except ClientError as e:
            logger.error(f"SQS publish failed for {source}: {e}")
            return None  # Don't fail the webhook — log and continue

    def _extract_event_type(self, payload: dict, source: str) -> str:
        if source == "jira":
            return payload.get("webhookEvent", "unknown")
        elif source == "slack":
            return payload.get("type", payload.get("event", {}).get("type", "unknown"))
        return payload.get("event_type", "unknown")

# Singleton for app lifecycle
from config import settings
sqs_producer = SQSProducer(
    queue_url=settings.SQS_QUEUE_URL,
    region=settings.AWS_REGION,
)
```

import pytest, json, boto3
from moto import mock_aws
from unittest.mock import patch
from services.sqs_producer import SQSProducer

@pytest.fixture
def sqs_queue(aws_credentials):
    with mock_aws():
        client = boto3.client("sqs", region_name="ap-southeast-1")
        queue = client.create_queue(QueueName="test-swarmforge-tasks")
        yield client, queue["QueueUrl"]

@pytest.fixture
def aws_credentials(monkeypatch):
    monkeypatch.setenv("AWS_ACCESS_KEY_ID", "testing")
    monkeypatch.setenv("AWS_SECRET_ACCESS_KEY", "testing")
    monkeypatch.setenv("AWS_SECURITY_TOKEN", "testing")
    monkeypatch.setenv("AWS_SESSION_TOKEN", "testing")
    monkeypatch.setenv("AWS_DEFAULT_REGION", "ap-southeast-1")

def test_publish_jira_event(sqs_queue):
    client, queue_url = sqs_queue
    producer = SQSProducer(queue_url=queue_url, region="ap-southeast-1")
    payload = {
        "webhookEvent": "jira:issue_updated",
        "issue": {"id": "12345", "fields": {"labels": [{"name": "ai-task"}]}},
        "timestamp": 1700000000000,
    }
    msg_id = producer.publish(payload, "jira")
    assert msg_id is not None

    messages = client.receive_message(QueueUrl=queue_url, MaxNumberOfMessages=1)
    body = json.loads(messages["Messages"][0]["Body"])
    assert body["source"] == "jira"
    assert body["event_type"] == "jira:issue_updated"
    assert "task_id" in body
    assert "idempotency_key" in body

def test_idempotency_key_same_event():
    producer = SQSProducer(queue_url="http://test", region="test")
    payload1 = {"webhookEvent": "jira:issue_updated", "issue": {"id": "999"}, "timestamp": 1700001000000}
    payload2 = {"webhookEvent": "jira:issue_updated", "issue": {"id": "999"}, "timestamp": 1700002000000}
    # Same hour bucket → same key
    key1 = producer._build_idempotency_key(payload1, "jira")
    key2 = producer._build_idempotency_key(payload2, "jira")
    assert key1 == key2

def test_sqs_failure_does_not_raise(monkeypatch, sqs_queue):
    _, queue_url = sqs_queue
    producer = SQSProducer(queue_url="https://bad-url", region="ap-southeast-1")
    result = producer.publish({"webhookEvent": "test"}, "jira")
    assert result is None  # Should return None, not raise
